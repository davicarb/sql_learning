create database seguradora_tp

create table cliente (
	cd_cnpj varchar(14) primary key,
	nm_razao_social varchar(150)
);

create table seguradora (
	cd_cnpj varchar(14) primary key,
	nm_razao_social varchar(150)
);

create table plano (
	cd_plano int primary key,
	nm_plano varchar(350),
	sg_plano varchar(50) not null,
	ds_beneficios varchar(1000)
);

create table contrato (
	cd_contrato int primary key,
	dt_assinatura date DEFAULT getdate(),
	cd_cnpj_cliente varchar(14) foreign key references cliente,
	cd_cnpj_seguradora varchar(14) foreign key references seguradora
);

create table apolice (
	cd_apolice int primary key,
	dt_validade date,
	cd_plano int foreign key references plano,
	cd_contrato int foreign key references contrato,
	CHECK (dt_validade > getdate())
);

create table segurado (
	cd_segurado int primary key,
	nm_segurado varchar(100),
	dt_nascimento date,
	cd_apolice int foreign key references apolice,
	ic_titular bit,
	CHECK (dt_nascimento < getdate())
);
